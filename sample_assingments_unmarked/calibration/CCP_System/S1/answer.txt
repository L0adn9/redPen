===== Airport.java =====
import java.util.concurrent.Semaphore;
import java.util.concurrent.atomic.AtomicReferenceArray;

/**
 * Airport - the shared physical resources every thread competes for.
 *
 *  runway : ONE runway shared by landings and take-offs -> fair binary Semaphore (FIFO, no starvation).
 *  gates  : the 3 gates; slot holds the docked Plane or null. Changed only with compareAndSet, so two
 *           planes can never occupy the same gate even if a bug tried to put them there.
 *  truck  : the single refuelling truck (a thread of its own).
 */
final class Airport {
    final Semaphore runway = new Semaphore(1, true);
    final AtomicReferenceArray<Plane> gates = new AtomicReferenceArray<Plane>(SimConfig.GATES);
    final RefuelTruck truck = new RefuelTruck();
    final Stats stats = new Stats();

    int dockedGates() {
        int n = 0;
        for (int i = 0; i < gates.length(); i++) {
            if (gates.get(i) != null) {
                n++;
            }
        }
        return n;
    }

    boolean allGatesEmpty() { return dockedGates() == 0; }
}

import java.util.Random;

/**
 * Asia Pacific Airport - concurrent simulation (CT074-3-2 Individual Assignment).
 *
 * Threads in this program
 *   Main              : only wires things together and waits - it never acts for anybody else
 *   ATC               : air traffic controller (landing queue, ground permits, gates, final report)
 *   Generator         : creates a new plane every 0, 1 or 2 seconds (also stages the congestion scenario)
 *   Plane-N           : one pilot per aircraft
 *   Passenger-PN-xx   : one thread per passenger
 *   Supply-/Cleaning-Crew-PN, Refuel-Truck : ground services
 *   Watchdog          : deadlock / stall detector
 *   Logger-Writer     : serialises console output
 *
 * Run:  javac *.java   then   java AirportSimulation [randomSeed]
 */
public class AirportSimulation {

    public static void main(String[] args) throws InterruptedException {
        long seed = args.length > 0 ? Long.parseLong(args[0]) : 2026L;   // fixed default = reproducible demo; pass another number to vary the run
        SimLog.start();
        SimLog.log("=== Asia Pacific Airport simulation starting (random seed " + seed + ") ===");

        Airport airport = new Airport();
        Watchdog watchdog = new Watchdog();
        AirTrafficControl atc = new AirTrafficControl(airport, SimConfig.TOTAL_PLANES, watchdog);

        Thread watchdogThread = new Thread(watchdog, "Watchdog");
        watchdogThread.setDaemon(true);
        Thread truckThread = new Thread(airport.truck, "Refuel-Truck");
        Thread atcThread = new Thread(atc, "ATC");
        Thread generator = new Thread(new ArrivalGenerator(airport, atc, new Random(seed)), "Generator");

        watchdogThread.start();
        truckThread.start();
        atcThread.start();
        generator.start();

        atcThread.join();        // ATC ends once every plane has left and the report is printed
        generator.join();
        truckThread.join();
        watchdog.stop();
        SimLog.log("=== Simulation finished ===");
        SimLog.stop();
    }

    /**
     * ArrivalGenerator - a new plane every 0, 1 or 2 seconds. Plane number EMERGENCY_PLANE_NO is held
     * back until the congested situation of the brief really exists (2+ planes circling, 2+ gates
     * occupied, ground full) and then arrives with a fuel shortage.
     */
    private static final class ArrivalGenerator implements Runnable {
        private final Airport airport;
        private final AirTrafficControl atc;
        private final Random rng;

        ArrivalGenerator(Airport airport, AirTrafficControl atc, Random rng) {
            this.airport = airport;
            this.atc = atc;
            this.rng = rng;
        }

        public void run() {
            for (int no = 1; no <= SimConfig.TOTAL_PLANES; no++) {
                int fuel = 60 + rng.nextInt(41);                       // 60..100 %
                if (no == SimConfig.EMERGENCY_PLANE_NO) {
                    waitForCongestion();
                    fuel = SimConfig.EMERGENCY_FUEL_PERCENT;
                }
                int arriving = 10 + rng.nextInt(SimConfig.MAX_PASSENGERS - 9);   // 10..50
                int departing = 10 + rng.nextInt(SimConfig.MAX_PASSENGERS - 9);  // 10..50
                Plane plane = new Plane(no, airport, atc, fuel, arriving, departing);
                SimLog.log("Radar: new aircraft inbound -> Plane " + no);
                new Thread(plane, "Plane-" + no).start();
                if (no < SimConfig.TOTAL_PLANES) {
                    SimConfig.pause(rng.nextInt(3) * 1000L);           // next arrival in 0, 1 or 2 s
                }
            }
        }

        private void waitForCongestion() {
            long start = System.currentTimeMillis();
            long deadline = start + 20_000;
            while (System.currentTimeMillis() < deadline) {
                // First 8 s: insist on EXACTLY the brief's picture (2 circling, 2 gates occupied, ground full).
                // After that accept 2 or more occupied gates so the scenario can never stall the run.
                boolean strict = System.currentTimeMillis() - start < 8_000;
                int gates = airport.dockedGates();
                boolean gatesOk = strict ? gates == SimConfig.CONGESTION_GATES : gates >= SimConfig.CONGESTION_GATES;
                if (atc.circlingCount() >= SimConfig.CONGESTION_WAITING_MIN
                        && gatesOk
                        && atc.groundCount() >= SimConfig.GROUND_CAPACITY) {
                    SimLog.log("SCENARIO: congestion confirmed - " + atc.circlingCount() + " planes circling, "
                            + gates + " gates occupied, ground full. A plane with a fuel shortage now arrives!");
                    return;
                }
                SimConfig.pause(50);
            }
            SimLog.log("SCENARIO: congestion not reached within 20s - releasing the emergency plane anyway.");
        }
    }
}
===== AirTrafficControl.java =====
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.ReentrantLock;

/**
 * AirTrafficControl - the ATC manager thread ("ATC").
 *
 * Responsibilities
 *  - Keeps the circle (holding) queue of planes that asked to land.
 *  - Lets a plane land ONLY when fewer than GROUND_CAPACITY (3) aircraft are on the ground. This
 *    permit is the airport-wide invariant that stops collisions on runway/gates.
 *  - Gives MAYDAY (low-fuel) planes priority over everybody else in the circle.
 *  - Reserves a gate at clearance time, so a cleared plane never needs to wait for a gate
 *    (the airport has no waiting area on the ground).
 *  - At the end performs the sanity checks and prints the statistics.
 *
 * Concurrency design: ONE ReentrantLock (fair) guards all ATC state; the ATC thread sleeps on the
 * Condition 'changed' and is woken by a plane that requests landing, declares an emergency or
 * departs. Replies to pilots travel through a CompletableFuture, so the pilot thread blocks
 * without polling and the ATC thread never executes pilot code.
 */
final class AirTrafficControl implements Runnable {

    /** One plane's request to land, waiting in the circle. */
    private static final class Request {
        final Plane plane;
        final long seq;                 // arrival order (FIFO tie-breaker)
        final long requestedAtMs;
        boolean emergency;
        boolean acknowledged;           // has the ATC thread already announced the MAYDAY?
        final CompletableFuture<Integer> clearance = new CompletableFuture<Integer>();

        Request(Plane plane, long seq, boolean emergency) {
            this.plane = plane;
            this.seq = seq;
            this.emergency = emergency;
            this.requestedAtMs = System.currentTimeMillis();
        }
    }

    private final Airport airport;
    private final int totalPlanes;
    private final Watchdog watchdog;

    private final ReentrantLock lock = new ReentrantLock(true);
    private final Condition changed = lock.newCondition();

    // ---- all fields below are guarded by 'lock' ------------------------------------------------
    private final List<Request> circle = new ArrayList<Request>();
    private final boolean[] gateReserved = new boolean[SimConfig.GATES];
    private final List<Integer> departureNotices = new ArrayList<Integer>();
    private int groundCount;
    private int departed;
    private long nextSeq;
    private int lastBlockedSize = -1;

    AirTrafficControl(Airport airport, int totalPlanes, Watchdog watchdog) {
        this.airport = airport;
        this.totalPlanes = totalPlanes;
        this.watchdog = watchdog;
    }

    // ================================ called by PILOT threads ====================================

    /** A pilot joins the circle. Returns a future that completes with the assigned gate number. */
    CompletableFuture<Integer> requestLanding(Plane plane, boolean emergency) {
        lock.lock();
        try {
            Request r = new Request(plane, nextSeq++, emergency);
            circle.add(r);
            changed.signalAll();          // wake the ATC thread
            return r.clearance;
        } finally {
            lock.unlock();
        }
    }

    /** A circling pilot whose fuel dropped below the threshold escalates its request to MAYDAY. */
    void declareEmergency(Plane plane) {
        lock.lock();
        try {
            for (Request r : circle) {
                if (r.plane == plane) {
                    r.emergency = true;
                }
            }
            changed.signalAll();
        } finally {
            lock.unlock();
        }
    }

    /** The pilot leaves the gate: its gate may be given to the next plane. */
    void releaseGate(int gate) {
        lock.lock();
        try {
            gateReserved[gate - 1] = false;
        } finally {
            lock.unlock();
        }
    }

    /** The plane has finished take-off: its ground permit is returned. */
    void planeDeparted(Plane plane) {
        lock.lock();
        try {
            groundCount--;
            Safety.leave("ground");
            departed++;
            departureNotices.add(plane.number());
            changed.signalAll();          // a ground slot is free -> wake the ATC thread
        } finally {
            lock.unlock();
        }
    }

    // ================================ read-only snapshots =======================================

    int circlingCount() {
        lock.lock();
        try {
            return circle.size();
        } finally {
            lock.unlock();
        }
    }

    int groundCount() {
        lock.lock();
        try {
            return groundCount;
        } finally {
            lock.unlock();
        }
    }

    // ================================ the ATC thread ============================================

    public void run() {
        SimLog.log("Air traffic control online. Runway open. Ground capacity "
                + SimConfig.GROUND_CAPACITY + ", gates " + SimConfig.GATES + ".");
        lock.lock();
        try {
            while (true) {
                announceNotices();

                if (!circle.isEmpty() && groundCount < SimConfig.GROUND_CAPACITY) {
                    grantLanding(pickNext());
                    continue;                       // look for more work before sleeping
                }
                if (departed == totalPlanes && circle.isEmpty()) {
                    break;                          // everyone has left: end of the day
                }
                if (!circle.isEmpty() && circle.size() != lastBlockedSize) {
                    lastBlockedSize = circle.size();
                    SimLog.log("Ground is full (" + groundCount + "/" + SimConfig.GROUND_CAPACITY
                            + "). Holding in the circle: " + circleDescription());
                }
                try {
                    changed.await();                // sleep until something changes
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        } finally {
            lock.unlock();
        }
        finalReport();
        airport.truck.shutdown();
    }

    /** Called with the lock held: log MAYDAYs (once) and departures, using THIS thread's identity. */
    private void announceNotices() {
        for (Request r : circle) {
            if (r.emergency && !r.acknowledged) {
                r.acknowledged = true;
                SimLog.log("MAYDAY received from Plane " + r.plane.number()
                        + ". Plane " + r.plane.number() + " now has landing priority over all other traffic.");
            }
        }
        for (Integer no : departureNotices) {
            SimLog.log("Plane " + no + " has left our airspace. Ground " + groundCount + "/"
                    + SimConfig.GROUND_CAPACITY + ".");
        }
        departureNotices.clear();
    }

    /** Emergency first, then first-come-first-served. */
    private Request pickNext() {
        Request best = null;
        for (Request r : circle) {
            if (best == null
                    || (r.emergency && !best.emergency)
                    || (r.emergency == best.emergency && r.seq < best.seq)) {
                best = r;
            }
        }
        circle.remove(best);
        return best;
    }

    private void grantLanding(Request r) {
        groundCount++;
        Safety.enter("ground", SimConfig.GROUND_CAPACITY);
        int gate = reserveGate();
        long waited = System.currentTimeMillis() - r.requestedAtMs;
        airport.stats.recordWait(r.plane.number(), waited);
        if (r.emergency) {
            airport.stats.emergencyLanding();
        }
        lastBlockedSize = -1;
        SimLog.log("Plane " + r.plane.number() + ": cleared to land" + (r.emergency ? " [EMERGENCY PRIORITY]" : "")
                + ". Assigned Gate " + gate + ". Ground " + groundCount + "/" + SimConfig.GROUND_CAPACITY
                + ", still circling: " + circleDescription());
        r.clearance.complete(gate);                 // wakes the pilot thread
    }

    private int reserveGate() {
        for (int i = 0; i < gateReserved.length; i++) {
            if (!gateReserved[i]) {
                gateReserved[i] = true;
                return i + 1;
            }
        }
        // Cannot happen: at most 3 planes are on the ground and each holds at most one gate.
        Safety.violation("no free gate although ground count is " + groundCount);
        return 1;
    }

    private String circleDescription() {
        if (circle.isEmpty()) {
            return "none";
        }
        StringBuilder sb = new StringBuilder();
        for (Request r : circle) {
            if (sb.length() > 0) {
                sb.append(", ");
            }
            sb.append("Plane ").append(r.plane.number());
            if (r.emergency) {
                sb.append("(MAYDAY)");
            }
        }
        return sb.toString();
    }

    // ================================ sanity checks + statistics ================================

    private void finalReport() {
        Stats s = airport.stats;
        SimLog.log("All planes have left. Starting end-of-day sanity checks...");
        boolean ok = true;
        ok &= check("All gates are empty", airport.allGatesEmpty());
        ok &= check("No aircraft left on the ground (ground count = 0)", groundCount == 0);
        ok &= check("Runway is free (permit returned)", airport.runway.availablePermits() == 1);
        ok &= check("Refuel truck is idle", Safety.current("refuel-truck") == 0);
        ok &= check("Planes served = planes expected (" + s.planesServed() + "/" + totalPlanes + ")",
                s.planesServed() == totalPlanes);
        ok &= check("Passengers disembarked = expected (" + s.paxDisembarked() + "/" + s.paxDisembarkPlanned() + ")",
                s.paxDisembarked() == s.paxDisembarkPlanned());
        ok &= check("Passengers boarded = expected (" + s.paxBoarded() + "/" + s.paxBoardPlanned() + ")",
                s.paxBoarded() == s.paxBoardPlanned());
        ok &= check("Peak aircraft on ground <= " + SimConfig.GROUND_CAPACITY + " (peak " + Safety.peak("ground") + ")",
                Safety.peak("ground") <= SimConfig.GROUND_CAPACITY);
        ok &= check("Peak aircraft on runway <= 1 (peak " + Safety.peak("runway") + ")", Safety.peak("runway") <= 1);
        ok &= check("Peak refuel-truck users <= 1 (peak " + Safety.peak("refuel-truck") + ")",
                Safety.peak("refuel-truck") <= 1);
        long gatePeak = 0;
        for (int g = 1; g <= SimConfig.GATES; g++) {
            gatePeak = Math.max(gatePeak, Safety.peak("gate-" + g));
        }
        ok &= check("Peak planes at any single gate <= 1 (peak " + gatePeak + ")", gatePeak <= 1);
        ok &= check("Safety violations recorded = 0 (found " + Safety.violations() + ")", Safety.violations() == 0);
        ok &= check("Watchdog found no deadlock or stall", watchdog.isClean());
        SimLog.log("SANITY CHECK RESULT: " + (ok ? "ALL CHECKS PASSED" : "*** SOME CHECKS FAILED ***"));

        SimLog.log("================ STATISTICS ================");
        SimLog.log(String.format(Locale.ENGLISH, "Waiting time to land  - minimum: %.2fs | average: %.2fs | maximum: %.2fs",
                s.minWaitMs() / 1000.0, s.avgWaitMs() / 1000.0, s.maxWaitMs() / 1000.0));
        for (java.util.Map.Entry<Integer, Long> e : s.waitByPlane().entrySet()) {
            SimLog.log(String.format(Locale.ENGLISH, "   Plane %d waited %.2fs", e.getKey(), e.getValue() / 1000.0));
        }
        SimLog.log("Planes served        : " + s.planesServed());
        SimLog.log("Emergency landings   : " + s.emergencies());
        SimLog.log("Passengers boarded   : " + s.paxBoarded());
        SimLog.log("Passengers deplaned  : " + s.paxDisembarked());
        SimLog.log("============================================");
    }

    private boolean check(String what, boolean passed) {
        SimLog.log("  [" + (passed ? "PASS" : "FAIL") + "] " + what);
        return passed;
    }
}
===== GroundCrew.java =====
import java.util.concurrent.CountDownLatch;

/**
 * GroundCrew - the supply crew and the cleaning crew. Each is a separate thread per plane, so
 * "refill supplies", "clean the cabin" and (via the truck) "refuel" all proceed at the same time
 * on one aircraft, and across the three gates at once.
 */
final class GroundCrew implements Runnable {
    enum Job { SUPPLIES, CLEANING }

    private final Job job;
    private final Plane plane;
    private final int gate;
    private final CountDownLatch serviceDone;

    GroundCrew(Job job, Plane plane, int gate, CountDownLatch serviceDone) {
        this.job = job;
        this.plane = plane;
        this.gate = gate;
        this.serviceDone = serviceDone;
    }

    public void run() {
        String what = job == Job.SUPPLIES ? "Restocking food and supplies on" : "Cleaning the cabin of";
        SimLog.log(what + " Plane " + plane.number() + " at Gate " + gate + "...");
        SimConfig.pause(job == Job.SUPPLIES ? SimConfig.SUPPLIES : SimConfig.CLEANING);
        SimLog.log((job == Job.SUPPLIES ? "Supplies restocked" : "Cabin cleaned") + " for Plane " + plane.number() + ".");
        serviceDone.countDown();
    }
}
===== Passenger.java =====
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.Semaphore;

/**
 * Passenger - one thread per passenger. Many passengers (of several planes at different gates) walk
 * through the gate doors AT THE SAME TIME: that is the "concurrent embark/disembark" requirement.
 *
 * Seat accounting: a plane owns a Semaphore with one permit per FREE seat (max 50).
 *   disembark -> release()  (a seat becomes free)
 *   embark    -> tryAcquire() (a seat is taken; failure would mean overbooking = safety violation)
 */
final class Passenger implements Runnable {
    enum Kind { DISEMBARK, EMBARK }

    private final Kind kind;
    private final Plane plane;
    private final int gate;
    private final int index;     // 1..total
    private final int total;
    private final CountDownLatch done;

    Passenger(Kind kind, Plane plane, int gate, int index, int total, CountDownLatch done) {
        this.kind = kind;
        this.plane = plane;
        this.gate = gate;
        this.index = index;
        this.total = total;
        this.done = done;
    }

    public void run() {
        SimConfig.pause(SimConfig.PASSENGER);                     // walking through the door
        Semaphore seats = plane.seats();
        if (kind == Kind.DISEMBARK) {
            seats.release();
            plane.stats().passengerOff();
        } else {
            if (!seats.tryAcquire()) {
                Safety.violation("Plane " + plane.number() + " overbooked (passenger " + index + ")");
            }
            plane.stats().passengerOn();
        }
        if (SimConfig.LOG_EVERY_PASSENGER || index <= 2 || index == total) {
            SimLog.log("I'm " + (kind == Kind.DISEMBARK ? "getting off Plane " : "boarding Plane ")
                    + plane.number() + " at Gate " + gate + " (" + index + "/" + total + ")"
                    + (index == total ? " - last one" : ""));
        }
        done.countDown();                                          // tell the pilot one more is through
    }
}
===== Plane.java =====
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.Semaphore;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

/**
 * Plane - one thread per aircraft; the thread IS the pilot. It performs the whole life-cycle:
 *
 *   request landing -> circle until cleared -> land -> coast to gate -> dock -> passengers get off
 *   -> (refuel || restock || clean) -> passengers board -> undock -> coast to runway -> take off.
 *
 * Every log line a pilot prints is about the pilot's own plane; other actors (ATC, truck, crews,
 * passengers) print their own events from their own threads.
 */
final class Plane implements Runnable {
    private final int number;
    private final Airport airport;
    private final AirTrafficControl atc;
    private final int arrivingPassengers;
    private final int departingPassengers;
    private final Semaphore seats;                // one permit per FREE seat
    private volatile int fuel;                    // percent

    Plane(int number, Airport airport, AirTrafficControl atc, int fuel, int arriving, int departing) {
        this.number = number;
        this.airport = airport;
        this.atc = atc;
        this.fuel = fuel;
        this.arrivingPassengers = arriving;
        this.departingPassengers = departing;
        this.seats = new Semaphore(SimConfig.MAX_PASSENGERS - arriving);
        airport.stats.planned(arriving, departing);
    }

    int number()      { return number; }
    Semaphore seats() { return seats; }
    Stats stats()     { return airport.stats; }
    void fillTank()   { fuel = 100; }

    private void say(String msg) { SimLog.log("Plane " + number + ": " + msg); }

    public void run() {
        long arrivedAt = System.currentTimeMillis();

        // ---- 1. ask ATC for permission to land ---------------------------------------------------
        boolean mayday = fuel < SimConfig.LOW_FUEL_PERCENT;
        if (mayday) {
            say("MAYDAY MAYDAY MAYDAY! Fuel shortage (" + fuel + "%). Requesting EMERGENCY landing!");
        } else {
            say("Requesting permission to land! (fuel " + fuel + "%, " + arrivingPassengers + " passengers on board)");
        }
        CompletableFuture<Integer> clearance = atc.requestLanding(this, mayday);

        // ---- 2. circle until cleared, burning fuel; escalate to MAYDAY if it gets low -----------
        int gate = circleUntilCleared(clearance, mayday);

        // ---- 3. land (runway is exclusive) -----------------------------------------------------
        say("Cleared! Waiting for the runway to land...");
        useRunway("Touchdown! Landing on the runway.", "Landed, vacating the runway.", SimConfig.LAND);

        // ---- 4. coast to the assigned gate and dock -------------------------------------------
        say("Coasting to Gate " + gate + ".");
        SimConfig.pause(SimConfig.COAST);
        Safety.enter("gate-" + gate, 1);
        if (!airport.gates.compareAndSet(gate - 1, null, this)) {   // atomic: only succeeds if gate is empty
            Safety.violation("Gate " + gate + " already occupied when Plane " + number + " tried to dock");
        }
        SimConfig.pause(SimConfig.DOCK);
        say("Docked at Gate " + gate + ". Doors open.");

        // ---- 5. passengers get off (concurrent threads) -----------------------------------------
        runPassengers(Passenger.Kind.DISEMBARK, arrivingPassengers, gate);
        say("All " + arrivingPassengers + " passengers have disembarked.");

        // ---- 6. refuel (exclusive truck) || restock || clean : three activities at once ----------
        CountDownLatch services = new CountDownLatch(3);
        new Thread(new GroundCrew(GroundCrew.Job.SUPPLIES, this, gate, services), "Supply-Crew-P" + number).start();
        new Thread(new GroundCrew(GroundCrew.Job.CLEANING, this, gate, services), "Cleaning-Crew-P" + number).start();
        int ahead = airport.truck.submit(new RefuelTruck.Job(this, gate, services));
        say("Requested refuel truck" + (ahead > 0 ? " (" + ahead + " plane(s) ahead of me)" : "")
                + ". Crews are working on my plane.");
        awaitUninterruptibly(services);
        say("Refuel, supplies and cleaning complete.");

        // ---- 7. passengers board (concurrent threads) ---------------------------------------------
        runPassengers(Passenger.Kind.EMBARK, departingPassengers, gate);
        say("Boarding complete: " + departingPassengers + " passengers on board.");

        // ---- 8. undock, coast to the runway, take off ------------------------------------------
        SimConfig.pause(SimConfig.UNDOCK);
        airport.gates.compareAndSet(gate - 1, this, null);          // free the gate (atomic)
        Safety.leave("gate-" + gate);
        atc.releaseGate(gate);
        say("Undocked from Gate " + gate + ". Coasting to the runway.");
        SimConfig.pause(SimConfig.COAST);
        useRunway("Cleared for departure. Taking off!", "Airborne. Goodbye Asia Pacific Airport!", SimConfig.TAKEOFF);

        // ---- 9. hand the ground permit back --------------------------------------------------
        airport.stats.planeServed();
        atc.planeDeparted(this);
        say("Total time at/around the airport: "
                + String.format("%.1f", (System.currentTimeMillis() - arrivedAt) / 1000.0) + "s.");
    }

    /** Holding pattern: block on the future in short ticks, burning fuel, and call MAYDAY when low. */
    private int circleUntilCleared(CompletableFuture<Integer> clearance, boolean alreadyMayday) {
        boolean mayday = alreadyMayday;
        while (true) {
            try {
                return clearance.get((long) (SimConfig.HOLDING_TICK_MS * SimConfig.TIME_SCALE), TimeUnit.MILLISECONDS);
            } catch (TimeoutException tick) {
                fuel -= SimConfig.FUEL_BURN_PER_TICK;
                if (!mayday && fuel < SimConfig.LOW_FUEL_PERCENT) {
                    mayday = true;
                    say("Fuel low (" + fuel + "%) while circling! Declaring MAYDAY.");
                    atc.declareEmergency(this);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                throw new IllegalStateException("Pilot of Plane " + number + " interrupted", e);
            } catch (ExecutionException e) {
                throw new IllegalStateException(e);
            }
        }
    }

    /** Take the single runway (fair semaphore => FIFO), use it, and always give it back. */
    private void useRunway(String onEntry, String onExit, int[] duration) {
        airport.runway.acquireUninterruptibly();
        try {
            Safety.enter("runway", 1);
            say(onEntry);
            SimConfig.pause(duration);
            say(onExit);
        } finally {
            Safety.leave("runway");
            airport.runway.release();
        }
    }

    private void runPassengers(Passenger.Kind kind, int count, int gate) {
        CountDownLatch latch = new CountDownLatch(count);
        String tag = kind == Passenger.Kind.DISEMBARK ? "D" : "B";
        for (int i = 1; i <= count; i++) {
            new Thread(new Passenger(kind, this, gate, i, count, latch), "Passenger-P" + number + "-" + tag + i).start();
        }
        awaitUninterruptibly(latch);
    }

    private static void awaitUninterruptibly(CountDownLatch latch) {
        boolean interrupted = false;
        while (latch.getCount() > 0) {
            try {
                latch.await();
            } catch (InterruptedException e) {
                interrupted = true;
            }
        }
        if (interrupted) {
            Thread.currentThread().interrupt();
        }
    }
}
===== RefuelTruck.java =====
import java.util.concurrent.BlockingQueue;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.LinkedBlockingQueue;

/**
 * RefuelTruck - the airport's ONLY refuelling truck, modelled as its own thread.
 *
 * Pilots put refuel jobs on a BlockingQueue; the single truck thread serves them one at a time, in
 * order. Because there is exactly one consumer, refuelling is mutually exclusive by construction,
 * and the queue is FIFO so no plane is starved. Safety.enter("refuel-truck", 1) double-checks it.
 */
final class RefuelTruck implements Runnable {

    /** A request from a pilot. */
    static final class Job {
        final Plane plane;
        final int gate;
        final CountDownLatch serviceDone;
        Job(Plane plane, int gate, CountDownLatch serviceDone) {
            this.plane = plane;
            this.gate = gate;
            this.serviceDone = serviceDone;
        }
    }

    private static final Job POISON = new Job(null, 0, null);
    private final BlockingQueue<Job> jobs = new LinkedBlockingQueue<Job>();

    /** Called by a pilot; returns how many jobs were already waiting (for the log message). */
    int submit(Job job) {
        int ahead = jobs.size();
        jobs.add(job);
        return ahead;
    }

    void shutdown() { jobs.add(POISON); }

    public void run() {
        SimLog.log("Refuel truck ready. Only one aircraft can be refuelled at a time.");
        try {
            while (true) {
                Job job = jobs.take();                    // blocks while idle
                if (job == POISON) {
                    break;
                }
                Safety.enter("refuel-truck", 1);
                SimLog.log("Refuelling Plane " + job.plane.number() + " at Gate " + job.gate
                        + " (truck in exclusive use). Waiting in line: " + jobs.size());
                SimConfig.pause(SimConfig.REFUEL);
                job.plane.fillTank();
                SimLog.log("Plane " + job.plane.number() + " refuelled to 100%. Truck is free again.");
                Safety.leave("refuel-truck");
                job.serviceDone.countDown();
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }
}
===== Safety.java =====
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.LongAccumulator;

/**
 * Safety - run-time invariant checker ("black box" for the airport).
 *
 * Every critical zone (runway, ground, refuel truck, ...) reports enter()/leave() with its allowed
 * limit. The counters are lock-free atomics, so checking never slows or serialises the simulation.
 * If a limit is ever exceeded the breach is logged and counted; the ATC prints the totals in its
 * final sanity check, which turns "I think it is safe" into evidence.
 */
final class Safety {
    private Safety() { }

    private static final ConcurrentHashMap<String, AtomicInteger> CURRENT = new ConcurrentHashMap<String, AtomicInteger>();
    private static final ConcurrentHashMap<String, LongAccumulator> PEAK = new ConcurrentHashMap<String, LongAccumulator>();
    private static final AtomicInteger VIOLATIONS = new AtomicInteger();
    private static final AtomicLong PROGRESS = new AtomicLong();

    /** A thread enters a zone that may hold at most 'limit' occupants at once. */
    static void enter(String zone, int limit) {
        AtomicInteger c = CURRENT.computeIfAbsent(zone, k -> new AtomicInteger());
        LongAccumulator p = PEAK.computeIfAbsent(zone, k -> new LongAccumulator(Math::max, 0));
        int now = c.incrementAndGet();      // atomic read-modify-write: no lost updates
        p.accumulate(now);
        if (now > limit) {
            violation("zone '" + zone + "' holds " + now + " but the limit is " + limit);
        }
    }

    static void leave(String zone) {
        CURRENT.get(zone).decrementAndGet();
    }

    static void violation(String what) {
        VIOLATIONS.incrementAndGet();
        SimLog.log("*** SAFETY VIOLATION: " + what);
    }

    static int violations()           { return VIOLATIONS.get(); }
    static long peak(String zone)     { LongAccumulator p = PEAK.get(zone); return p == null ? 0 : p.get(); }
    static int current(String zone)   { AtomicInteger c = CURRENT.get(zone); return c == null ? 0 : c.get(); }
    static void progress()            { PROGRESS.incrementAndGet(); }
    static long progressCount()       { return PROGRESS.get(); }
}

===== SimLog.java =====
import java.util.concurrent.BlockingQueue;
import java.util.concurrent.LinkedBlockingQueue;

/**
 * SimLog - asynchronous console logger (producer/consumer).
 *
 * Every thread calls log(); the line is stamped with the NAME OF THE CALLING THREAD and the
 * simulation clock, then handed to one dedicated writer thread through a BlockingQueue.
 * Result: lines never interleave mid-sentence, and the output proves WHICH thread did WHAT
 * (the brief forbids one thread printing on behalf of another).
 */
final class SimLog {
    private SimLog() { }

    private static final BlockingQueue<String> QUEUE = new LinkedBlockingQueue<String>();
    private static final String POISON = "<<end-of-log>>";
    private static final long START = System.nanoTime();
    private static Thread writer;

    static void start() {
        writer = new Thread(new Runnable() {
            public void run() {
                try {
                    while (true) {
                        String line = QUEUE.take();
                        if (POISON.equals(line)) {
                            return;
                        }
                        System.out.println(line);
                    }
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }
        }, "Logger-Writer");
        writer.start();
    }

    /** Thread-safe: stamping + enqueue are done together so timestamps stay in order. */
    static void log(String message) {
        Safety.progress(); // every log line counts as "the system is alive" for the watchdog
        synchronized (QUEUE) {
            double t = (System.nanoTime() - START) / 1e9;
            QUEUE.add(String.format("[%6.2fs] %-20s: %s", t, Thread.currentThread().getName(), message));
        }
    }

    /** Flush everything and stop the writer. */
    static void stop() throws InterruptedException {
        QUEUE.add(POISON);
        writer.join();
    }
}

final class Stats {

    private final LongAdder waitSamples    = new LongAdder();

    Map<Integer, Long> waitByPlane() { return waitByPlane; }
}
===== Watchdog.java =====
import java.lang.management.ManagementFactory;
import java.lang.management.ThreadMXBean;

/**
 * Watchdog - daemon thread that guards against the two classic liveness failures:
 *   1. Deadlock  : asks the JVM (ThreadMXBean) whether any threads wait on each other in a cycle.
 *   2. Stall     : if no thread has logged anything for WATCHDOG_STALL_MS the system is stuck.
 * Either finding is logged and remembered so the ATC can report it in the sanity checks.
 */
final class Watchdog implements Runnable {
    private volatile boolean stopped;
    private volatile boolean problemFound;

    public void run() {
        ThreadMXBean mx = ManagementFactory.getThreadMXBean();
        long lastProgress = -1;
        long lastChange = System.currentTimeMillis();
        while (!stopped) {
            try {
                Thread.sleep(500);
            } catch (InterruptedException e) {
                break;
            }
            long[] deadlocked = mx.findDeadlockedThreads();
            if (deadlocked != null) {
                problemFound = true;
                SimLog.log("WATCHDOG: DEADLOCK among " + deadlocked.length + " threads!");
            }
            long p = Safety.progressCount();
            if (p != lastProgress) {
                lastProgress = p;
                lastChange = System.currentTimeMillis();
            } else if (System.currentTimeMillis() - lastChange > SimConfig.WATCHDOG_STALL_MS) {
                problemFound = true;
                lastChange = System.currentTimeMillis();
                SimLog.log("WATCHDOG: no progress for " + SimConfig.WATCHDOG_STALL_MS / 1000 + "s - possible stall!");
            }
        }
    }

    boolean isClean() { return !problemFound; }

    void stop() { stopped = true; }
}
